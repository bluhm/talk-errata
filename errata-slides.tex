\subsection{Application Binary Interface}
\begin{frame}{Application Binary Interface}
\begin{itemize}
\item library API and ABI constant
\item no changes to installed header files
\item function prototypes
\item data structs
\item \texttt{/usr/src/lib/check\_sym}
\item avoid C++
\end{itemize}
\end{frame}

\subsection{Hidden Dependencies}
\begin{frame}{Hidden Dependencies}
\begin{itemize}
\item statically linked \texttt{/bin} and \texttt{/sbin}
\item \texttt{/usr/src/sbin/unwind} copies libunbound
\end{itemize}
\end{frame}

\section{Ecosystem}

\subsection{Embargo Drawbacks}
\begin{frame}{Embargo Drawbacks}
\begin{itemize}
\item up to half a year
\item slow down development
\item no open commits and review
\item no community testing
\item NSA knows everything
\item AI re-discovers faster
\end{itemize}
\end{frame}

\subsection{Project or Company}
\begin{frame}{Project or Company}
Project
\begin{itemize}
\item OpenBSD can be very fast
\item hours from report to release
\item testing stable is hard
\item interaction with community
\end{itemize}
Company
\begin{itemize}
\item quality assurance takes time
\item long maintenance releases
\item weekends and holidays
\item should share fixes
\end{itemize}
\end{frame}
